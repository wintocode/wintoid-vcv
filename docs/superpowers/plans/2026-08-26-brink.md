**Goal:** Add Brink, a 12 HP polyphonic dual-window utility with bipolar position, directional boundary triggers, and shared logic, to the wintoid VCV Rack plugin.

**Architecture:** Put all voltage maths, hysteretic state, event timing, logic state, and polyphonic indexing in a C++11 header with no Rack dependency. Keep the Rack module as a thin adapter that resolves normalled ports and writes voltages, while a deterministic Python generator owns the SVG background and layout constants. Follow the existing one-directory-per-module and standalone-test conventions.

**Tech Stack:** C++11, VCV Rack 2 API, NanoVG, Python 3 standard library, SVG, Make, address/undefined-behaviour sanitizers.

**Spec:** `docs/superpowers/specs/2026-08-26-brink-design.md`

Create `src/Brink/dsp.h` with this public shape and the exact transition order from the spec:

```cpp
#pragma once

#include <cmath>

namespace brink {

static const float MIN_WIDTH = 0.001f;
static const float MAX_WIDTH = 20.f;
static const float HYSTERESIS = 0.001f;
static const int MAX_CHANNELS = 16;

enum Region { BELOW = 0, INSIDE, ABOVE };

struct WindowFrame {
    float signal;
    float center;
    float width;
    float lower;
    float upper;
    float relative;
    float position;
};

inline float sanitize(float value, float fallback)
{
    return std::isfinite(value) ? value : fallback;
}

inline float clampf(float value, float low, float high)
{
    return value < low ? low : (value > high ? high : value);
}

inline WindowFrame make_window(float signal, float center, float rawWidth)
{
    WindowFrame f;
    f.signal = sanitize(signal, 0.f);
    f.center = sanitize(center, 0.f);
    f.width = clampf(sanitize(rawWidth, MIN_WIDTH), MIN_WIDTH, MAX_WIDTH);
    f.lower = f.center - 0.5f * f.width;
    f.upper = f.center + 0.5f * f.width;
    f.relative = 10.f * (f.signal - f.center) / f.width;
    f.position = clampf(f.relative, -5.f, 5.f);
    return f;
}

inline Region classify_initial(const WindowFrame& f)
{
    if (f.signal < f.lower) return BELOW;
    if (f.signal > f.upper) return ABOVE;
    return INSIDE;
}

inline Region advance_region(Region previous, const WindowFrame& f)
{
    if (previous == BELOW) {
        if (f.signal >= f.upper + HYSTERESIS) return ABOVE;
        if (f.signal >= f.lower + HYSTERESIS) return INSIDE;
        return BELOW;
    }
    if (previous == INSIDE) {
        if (f.signal < f.lower - HYSTERESIS) return BELOW;
        if (f.signal > f.upper + HYSTERESIS) return ABOVE;
        return INSIDE;
    }
    if (f.signal <= f.lower - HYSTERESIS) return BELOW;
    if (f.signal <= f.upper - HYSTERESIS) return INSIDE;
    return ABOVE;
}

} // namespace brink
```

Add these public types and functions to `dsp.h`:

```cpp
enum EventId { LOW_UP = 0, HIGH_UP, LOW_DOWN, HIGH_DOWN, EVENT_COUNT };

struct WindowState {
    bool initialized;
    Region region;
    float pulseRemaining[EVENT_COUNT];

    WindowState() : initialized(false), region(INSIDE) {
        for (int i = 0; i < EVENT_COUNT; ++i) pulseRemaining[i] = 0.f;
    }
};

struct WindowOutput {
    Region region;
    bool inside;
    float position;
    bool eventHigh[EVENT_COUNT];
};

inline void reset(WindowState& state)
{
    state = WindowState();
}
```

Implement `process_window()` with this order:

1. Sanitize `sampleTime` to a non-negative finite value.
2. Create the current `WindowFrame`.
3. On the first sample, classify and mark initialized without starting pulses.
4. Otherwise advance the region and start the mapped pulses. Direct `BELOW -> ABOVE` starts `LOW_UP` and `HIGH_UP`; direct `ABOVE -> BELOW` starts `HIGH_DOWN` and `LOW_DOWN`.
5. Set each `eventHigh[i]` from `pulseRemaining[i] > 0`, then subtract `sampleTime` with a floor of zero.
6. Return `inside = region == INSIDE` and the frame's clamped position.

Starting or re-triggering an event assigns exactly `0.001f` seconds to its timer. Do not accumulate duration.

inline int effective_channels(int channels)
{
    if (channels < 1) return 1;
    return channels > MAX_CHANNELS ? MAX_CHANNELS : channels;
}

inline int logic_channels(int aChannels, int bChannels)
{
    int a = effective_channels(aChannels);
    int b = effective_channels(bChannels);
    return a > b ? a : b;
}

inline int broadcast_lane(int lane, int channels)
{
    int count = effective_channels(channels);
    if (count == 1 || lane >= count) return 0;
    return lane;
}
